# HadoopDelegationToken -> HADOOP_TOKEN_FILE_LOCATION=<path>/hadoop.token). Keep in
# sync with application.yaml.

--conf
spark.kubernetes.driver.secrets.gcp-secret=/etc/gcp
--conf
spark.kubernetes.driverEnv.GOOGLE_APPLICATION_CREDENTIALS=/etc/gcp/key.json
--conf
spark.kubernetes.driver.secrets.hadoop-secret=/etc/hadoop-token
--conf
